class User<T, K extends number>{
    constructor(public name: T, public age: K){}

    public getPsw(): string{
        return `${this.name}${this.age}`;
    }

    public getSecret(): number{
        return this.age * 2;
    }
}

const obj1 = new User("Igor", 25.6);
const obj2 = new User(123, 456);
const obj3 = new User("Irina", 456);
const obj4 = new User(true, 5);

console.log(obj1.getSecret());
console.log(obj2.getSecret());
console.log(obj3.getSecret());
console.log(obj4.getSecret());
